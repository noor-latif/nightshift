## Step-by-step reasoning

1. **Root cause:** `PasteHandler._read_body()` calls `int()` on the `Content-Length` header unguarded. A present-but-non-numeric value (`abc`) raises `ValueError` inside `do_POST` *outside* the existing `try/except`, so `BaseHTTPRequestHandler` gets an unhandled exception → broken connection instead of a JSON 400.

2. **Fix placement (criterion 2):** The guard must live where the header is parsed, inside `_read_body()`, so *any* caller benefits. The cleanest single-guard fix: wrap the `int()` parse in `try/except ValueError` and re-raise `ValueError` with a clear message. `do_POST` already catches `(ValueError, UnicodeDecodeError)` generically around `_read_body()` and answers `_send_json(400, {"error": "invalid JSON body"})` — so the malformed-header case now flows through the *existing* client-error path and gets exactly the established error convention: 400, `{"error": "..."}`, `Content-Type: application/json`. No special-casing of `do_POST`, no double-send, minimal diff.

3. **What I deliberately do NOT touch (criterion 5):** `GET /health`, `GET /paste/<id>`, `PasteStore`, `current_revision()`, port/revision semantics — untouched.

4. **New test file (criterion 3):** `content_length_test.py` at repo root, stdlib `unittest`, using raw `http.client` requests (urllib would auto-set a valid `Content-Length`, so raw sockets are needed to send `Content-Length: abc`). Covers: non-numeric header → 400 JSON error with non-empty `error` and JSON content type; connection stays usable afterwards; missing header regression → still 400 `{"error": "invalid JSON body"}`; valid header + `{"content": "hello"}` → 201 and round-trip GET 200.

5. **Test run:** `python3 -m unittest discover` from repo root → 17 existing + 4 new = **21 tests, OK**. No existing test modified or deleted.

+"""Tests for malformed Content-Length handling on POST /paste."""
+from __future__ import annotations
+
+import http.client
+import json
+import threading
+import unittest
+import urllib.request
+
+import app
+
+
+class ContentLengthHeaderTest(unittest.TestCase):
+    def setUp(self) -> None:
+        self.server = app.make_server(0)
+        self.port = self.server.server_address[1]
+        threading.Thread(target=self.server.serve_forever, daemon=True).start()
+        self.addCleanup(self.server.shutdown)
+        self.addCleanup(self.server.server_close)
+
+    def _post_raw(
+        self, content_length: str | None, body: bytes | None
+    ) -> tuple[int, str | None, bytes]:
+        conn = http.client.HTTPConnection("127.0.0.1", self.port, timeout=10)
+        self.addCleanup(conn.close)
+        conn.putrequest("POST", "/paste")
+        if content_length is not None:
+            conn.putheader("Content-Length", content_length)
+        conn.endheaders()
+        if body:
+            conn.send(body)
+        resp = conn.getresponse()
+        return resp.status, resp.getheader("Content-Type"), resp.read()
+
+    def test_non_numeric_content_length_gets_400_json_error(self) -> None:
+        status, ctype, body = self._post_raw("abc", b"x=1")
+        self.assertEqual(status, 400)
+        self.assertTrue(ctype and ctype.startswith("application/json"))
+        payload = json.loads(body)
+        self.assertIn("error", payload)
+        self.assertTrue(payload["error"])
+
+    def test_server_survives_malformed_header_request(self) -> None:
+        status, _, _ = self._post_raw("abc", b"x=1")
+        self.assertEqual(status, 400)
+        req = urllib.request.Request(
+            f"http://127.0.0.1:{self.port}/paste",
+            data=json.dumps({"content": "after"}).encode("utf-8"),
+            method="POST",
+        )
+        with urllib.request.urlopen(req) as resp:
+            self.assertEqual(resp.status, 201)
+
+    def test_missing_content_length_unchanged_invalid_json_body(self) -> None:
+        status, _, body = self._post_raw(None, None)
+        self.assertEqual(status, 400)
+        self.assertEqual(json.loads(body), {"error": "invalid JSON body"})
+
+    def test_valid_content_length_round_trip_unchanged(self) -> None:
+        payload = json.dumps({"content": "hello"}).encode("utf-8")
+        status, _, body = self._post_raw(str(len(payload)), payload)
+        self.assertEqual(status, 201)
+        paste_id = json.loads(body)["id"]
+        with urllib.request.urlopen(
+            f"http://127.0.0.1:{self.port}/paste/{paste_id}"
+        ) as resp:
+            self.assertEqual(resp.status, 200)
+            self.assertEqual(json.loads(resp.read()), {"content": "hello"})
+
+
+if __name__ == "__main__":
+    unittest.main()
